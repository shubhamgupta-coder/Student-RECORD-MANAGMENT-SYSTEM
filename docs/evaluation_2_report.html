2. Relational Model & Referential Integrity

In Version 2, the relational model expands from standalone entity tables into an interconnected relational schema:

  • STUDENT (1) — ENROLLMENT (N): A student can register for multiple academic terms. Foreign key check guarantees non-existent students cannot be enrolled.
  • COURSE (1) — ENROLLMENT (N): Courses contain multiple term enrollments. Foreign key check guarantees students are only enrolled in valid course codes.
  • STUDENT (1) — MARKS (N): Marks entries strictly require an existing parent student ID.
  • CHECK Constraint: $0 \le \text{score} \le \text{max\_score}$ enforced defensively before persistence.

#include "../include/Mark.h"

Mark::Mark() : markId(""), studentId(""), subject(""), assessment(""), score(0.0), maxScore(100.0) {}

Mark::Mark(const std::string& mId, const std::string& sId, const std::string& sub,
           const std::string& assess, double sc, double maxSc)
    : markId(mId), studentId(sId), subject(sub), assessment(assess), score(sc), maxScore(maxSc) {}

std::string Mark::getMarkId() const { return markId; }
std::string Mark::getStudentId() const { return studentId; }
std::string Mark::getSubject() const { return subject; }
std::string Mark::getAssessment() const { return assessment; }
double Mark::getScore() const { return score; }
double Mark::getMaxScore() const { return maxScore; }

bool Mark::validateScore(std::string& errorMessage) const {
    if (Utils::trim(markId).empty()) {
        errorMessage = "Validation Error: Mark ID cannot be empty.";
        return false;
    }
    if (Utils::trim(studentId).empty()) {
        errorMessage = "Validation Error: Student ID cannot be empty.";
        return false;
    }
    if (Utils::trim(subject).empty()) {
        errorMessage = "Validation Error: Subject cannot be empty.";
        return false;
    }
    if (maxScore <= 0.0) {
        errorMessage = "Validation Error: Maximum score must be greater than zero.";
        return false;
    }
    // TC-06: 0 <= score <= maxScore
    if (score < 0.0) {
        errorMessage = "Validation Error: Score cannot be negative.";
        return false;
    }
    if (score > maxScore) {
        std::ostringstream oss;
        oss << "Validation Error: Score (" << score << ") exceeds maximum allowed score (" << maxScore << ").";
        errorMessage = oss.str();
        return false;
    }
    return true;
}

double Mark::calculatePercentage() const {
    if (maxScore <= 0.0) return 0.0;
    return (score / maxScore) * 100.0;
}

std::string Mark::serialize() const {
    return markId + "|" + studentId + "|" + subject + "|" + assessment + "|" +
           std::to_string(score) + "|" + std::to_string(maxScore);
}

Mark Mark::deserialize(const std::string& line) {
    std::vector<std::string> parts = Utils::split(line, '|');
    if (parts.size() >= 6) {
        double sc = 0.0;
        double mx = 100.0;
        try {
            sc = std::stod(parts[4]);
            mx = std::stod(parts[5]);
        } catch (...) {}
        return Mark(parts[0], parts[1], parts[2], parts[3], sc, mx);
    }
    return Mark();
}

// Referential integrity enforcement in Version 2:
bool StudentRepository::addEnrollment(const Enrollment& enrollment, std::string& errorMessage) {
    // Foreign Key Check: Student must exist
    Student* student = findById(enrollment.getStudentId());
    if (!student) {
        errorMessage = "Foreign Key Error: Student ID '" + enrollment.getStudentId() + "' does not exist.";
        return false;
    }

    // Foreign Key Check: Course must exist
    Course* course = findCourseById(enrollment.getCourseId());
    if (!course) {
        errorMessage = "Foreign Key Error: Course ID '" + enrollment.getCourseId() + "' does not exist.";
        return false;
    }

    enrollmentsCache.push_back(enrollment);
    saveEnrollments();
    return true;
}

bool StudentRepository::addMark(const Mark& mark, std::string& errorMessage) {
    // Score Range Validation (TC-06)
    if (!mark.validateScore(errorMessage)) {
        return false;
    }

    // Foreign Key Check: Student must exist
    Student* student = findById(mark.getStudentId());
    if (!student) {
        errorMessage = "Foreign Key Error: Cannot enter mark. Student ID '" + mark.getStudentId() + "' does not exist.";
        return false;
    }

    marksCache.push_back(mark);
    saveMarks();
    return true;
}

std::vector<Mark> StudentRepository::getMarksByStudentId(const std::string& studentId) {
    std::vector<Mark> results;
    std::string upperId = Utils::toUpper(Utils::trim(studentId));
    for (const auto& m : marksCache) {
        if (Utils::toUpper(m.getStudentId()) == upperId) {
            results.push_back(m);
        }
    }
    return results;
}
