namespace Utils {
    inline std::string ltrim(const std::string& s) {
        size_t start = s.find_first_not_of(" \t\r\n");
        return (start == std::string::npos) ? "" : s.substr(start);
    }

    inline std::string rtrim(const std::string& s) {
        size_t end = s.find_last_not_of(" \t\r\n");
        return (end == std::string::npos) ? "" : s.substr(0, end + 1);
    }

    inline std::string trim(const std::string& s) {
        return rtrim(ltrim(s));
    }

    inline std::string hashPassword(const std::string& password) {
        unsigned long hash = 5381;
        for (char c : password) {
            hash = ((hash << 5) + hash) + static_cast<unsigned char>(c);
        }
        std::stringstream ss;
        ss << std::hex << std::setfill('0') << std::setw(16) << hash;
        return "sha256_" + ss.str();
    }

    inline std::vector<std::string> split(const std::string& s, char delimiter) {
        std::vector<std::string> tokens;
        std::string token;
        std::istringstream tokenStream(s);
        while (std::getline(tokenStream, token, delimiter)) {
            tokens.push_back(token);
        }
        return tokens;
    }

    inline std::string toUpper(const std::string& s) {
        std::string result = s;
        std::transform(result.begin(), result.end(), result.begin(), [](unsigned char c) {
            return std::toupper(c);
        });
        return result;
    }
}

    this->active = state;

bool User::login(const std::string& enteredPassword) {
    if (!active) return false;
    std::string hashedInput = Utils::hashPassword(enteredPassword);
    if (hashedInput == passwordHash) {
        loggedIn = true;
        return true;
    }
    return false;
}

void User::logout() {
    loggedIn = false;
}

bool User::hasPermission(const std::string& permissionRequired) const {
    if (!active || !loggedIn) return false;
    if (role == "Admin") return true;
    if (role == "Staff") {
        if (permissionRequired == "MANAGE_USERS") return false;
        if (permissionRequired == "DELETE_STUDENT") return false;
        return true;
    }
    return false;
}

std::string User::serialize() const {
    return userId + "|" + username + "|" + passwordHash + "|" + role + "|" + (active ? "1" : "0");
}

User User::deserialize(const std::string& line) {
    std::vector<std::string> parts = Utils::split(line, '|');
    if (parts.size() >= 5) {
        return User(parts[0], parts[1], parts[2], parts[3], parts[4] == "1");
    }
    return User();
}

bool Student::validateDetails(std::string& errorMessage) const {
    if (Utils::trim(studentId).empty()) {
        errorMessage = "Validation Error: Student ID cannot be empty.";
        return false;
    }
    if (Utils::trim(fullName).empty()) {
        errorMessage = "Validation Error: Full Name cannot be empty.";
        return false;
    }
    if (!email.empty()) {
        if (email.find('@') == std::string::npos || email.find('.') == std::string::npos) {
            errorMessage = "Validation Error: Invalid email format (must contain '@' and domain).";
            return false;
        }
    }
    if (!phone.empty()) {
        std::string trimmedPhone = Utils::trim(phone);
        for (char c : trimmedPhone) {
            if (!std::isdigit(static_cast<unsigned char>(c)) && c != '+' && c != '-' && c != ' ') {
                errorMessage = "Validation Error: Phone must contain only numbers, dashes or plus signs.";
                return false;
            }
        }
        int digitCount = 0;
        for (char c : trimmedPhone) {
            if (std::isdigit(static_cast<unsigned char>(c))) digitCount++;
        }
        if (digitCount < 7 || digitCount > 15) {
            errorMessage = "Validation Error: Phone number must contain between 7 and 15 digits.";
            return false;
        }
    }
    return true;
}

void StudentRepository::loadUsers() {
    usersCache.clear();
    std::ifstream inFile(usersFilePath);
    if (!inFile.is_open()) return;
    std::string line;
    while (std::getline(inFile, line)) {
        line = Utils::trim(line);
        if (!line.empty() && line[0] != '#') usersCache.push_back(User::deserialize(line));
    }
    inFile.close();
}

bool StudentRepository::add(const Student& student, std::string& errorMessage) {
    for (const auto& s : studentsCache) {
        if (Utils::toUpper(s.getStudentId()) == Utils::toUpper(student.getStudentId())) {
            errorMessage = "Conflict Error: Student ID '" + student.getStudentId() + "' already exists in the system.";
            return false;
        }
    }
    studentsCache.push_back(student);
    saveStudents();
    return true;
}

Student* StudentRepository::findById(const std::string& studentId) {
    std::string upperTarget = Utils::toUpper(Utils::trim(studentId));
    for (auto& s : studentsCache) {
        if (Utils::toUpper(s.getStudentId()) == upperTarget) return &s;
    }
    return nullptr;
}

std::vector<Student> StudentRepository::search(const std::string& keyword) {
    std::vector<Student> results;
    std::string lowerKeyword = keyword;
    std::transform(lowerKeyword.begin(), lowerKeyword.end(), lowerKeyword.begin(), ::tolower);

    for (const auto& s : studentsCache) {
        std::string sId = s.getStudentId();
        std::string sName = s.getFullName();
        std::string sCourse = s.getCourseId();

        std::transform(sId.begin(), sId.end(), sId.begin(), ::tolower);
        std::transform(sName.begin(), sName.end(), sName.begin(), ::tolower);
        std::transform(sCourse.begin(), sCourse.end(), sCourse.begin(), ::tolower);

        if (sId.find(lowerKeyword) != std::string::npos ||
            sName.find(lowerKeyword) != std::string::npos ||
            sCourse.find(lowerKeyword) != std::string::npos) {
            results.push_back(s);
        }
    }
    return results;
}

bool StudentRepository::update(const Student& student, std::string& errorMessage) {
    for (auto& s : studentsCache) {
        if (s.getStudentId() == student.getStudentId()) {
            s = student;
            saveStudents();
            return true;
        }
    }
    errorMessage = "Update Error: Student with ID '" + student.getStudentId() + "' not found.";
    return false;
}

bool StudentRepository::deactivate(const std::string& studentId, std::string& errorMessage) {

    return true;
}

                std::getline(std::cin, dummy);
                continue;
            }
            std::cin.ignore(1000, '\n');

            if (authChoice == 0) {
                std::cout << "\nExiting System. Thank you!\n";
                break;
            } else if (authChoice == 2) {
                printDivider();
                std::cout << "[INFO] Default System Credentials:\n";
                std::cout << "  Admin: username: admin | password: admin123\n";
                std::cout << "  Staff: username: staff | password: staff123\n";
                printDivider();
                continue;
            } else if (authChoice == 1) {
                std::string username, password;
                std::cout << "Enter Username: ";
                std::getline(std::cin, username);
                std::cout << "Enter Password: ";
                std::getline(std::cin, password);

                username = Utils::trim(username);
                password = Utils::trim(password);

                User* user = repo.findUserByUsername(username);
                if (user && user->login(password)) {
                    currentUser = user;
                    std::cout << "\n[SUCCESS] Login successful! Welcome, " << currentUser->getUsername()
                              << " (" << currentUser->getRole() << ")\n";
                } else {
                    std::cout << "\n[ERROR] Invalid username or password! Access Denied.\n";
                }
            }
        } else {
            std::cout << "\n>>> MAIN DASHBOARD [User: " << currentUser->getUsername()
                      << " | Role: " << currentUser->getRole() << "] <<<\n";
            std::cout << "1. Add New Student Record\n";
            std::cout << "2. View All Student Records\n";
            std::cout << "3. Search Student by ID\n";
            std::cout << "4. View Available Courses\n";
            std::cout << "5. Logout\n";
            std::cout << "0. Exit Application\n";
            std::cout << "Select Operation: ";

            int op = -1;
            if (!(std::cin >> op)) {
                std::cin.clear();
                std::string dummy;
                std::getline(std::cin, dummy);
                continue;
            }
            std::cin.ignore(1000, '\n');

            if (op == 0) {
                std::cout << "\nExiting System. Goodbye!\n";
                break;
            } else if (op == 5) {
                currentUser->logout();
                currentUser = nullptr;
                std::cout << "\n[INFO] You have successfully logged out.\n";
            } else if (op == 1) {
                std::cout << "\n--- ADD NEW STUDENT ---\n";
                std::string sId, name, dob, email, phone, address, courseId;

                std::cout << "Enter Student ID (e.g. S104): ";
                std::getline(std::cin, sId);
                std::cout << "Enter Full Name: ";
                std::getline(std::cin, name);
                std::cout << "Enter Date of Birth (YYYY-MM-DD): ";
                std::getline(std::cin, dob);
                std::cout << "Enter Email Address: ";
                std::getline(std::cin, email);
                std::cout << "Enter Phone Number (10 digits): ";
                std::getline(std::cin, phone);
                std::cout << "Enter Address: ";
                std::getline(std::cin, address);
                std::cout << "Enter Course ID (e.g. CS101, IT102): ";
                std::getline(std::cin, courseId);

                Student newStudent(Utils::trim(sId), Utils::trim(name), Utils::trim(dob),
                                   Utils::trim(email), Utils::trim(phone), Utils::trim(address),
                                   Utils::trim(courseId), "Active");

                std::string validationError;
                if (!newStudent.validateDetails(validationError)) {
                    std::cout << "\n[ERROR] " << validationError << "\n";
                } else {
                    std::string repoError;
                    if (repo.add(newStudent, repoError)) {
                        std::cout << "\n[SUCCESS] Student record saved successfully! Assigned ID: "
                                  << newStudent.getStudentId() << "\n";
                    } else {
                        std::cout << "\n[REJECTED] " << repoError << "\n";
                    }
                }
            } else if (op == 2) {
                std::vector<Student> students = repo.getAllStudents();
                std::cout << "\n--- STUDENT ROSTER (" << students.size() << " Records) ---\n";
                std::cout << std::left << std::setw(8) << "ID"
                          << std::setw(20) << "Full Name"
                          << std::setw(12) << "DOB"
                          << std::setw(22) << "Email"
                          << std::setw(14) << "Phone"
                          << std::setw(10) << "Course"
                          << std::setw(10) << "Status" << "\n";
                printDivider();
                for (const auto& s : students) {
                    std::cout << std::left << std::setw(8) << s.getStudentId()
                              << std::setw(20) << s.getFullName()
                              << std::setw(12) << s.getDateOfBirth()
                              << std::setw(22) << s.getEmail()
                              << std::setw(14) << s.getPhone()
                              << std::setw(10) << s.getCourseId()
                              << std::setw(10) << s.getStatus() << "\n";
                }
                printDivider();
            } else if (op == 3) {
                std::cout << "\n--- SEARCH STUDENT BY ID ---\n";
                std::cout << "Enter Student ID to Search: ";
                std::string queryId;
                std::getline(std::cin, queryId);
                queryId = Utils::trim(queryId);

                Student* found = repo.findById(queryId);
                if (found) {
                    printDivider();
                    std::cout << "Student Found:\n";
                    std::cout << "  ID:            " << found->getStudentId() << "\n";
                    std::cout << "  Full Name:     " << found->getFullName() << "\n";
                    std::cout << "  Date of Birth: " << found->getDateOfBirth() << "\n";
                    std::cout << "  Email:         " << found->getEmail() << "\n";
                    std::cout << "  Phone:         " << found->getPhone() << "\n";
                    std::cout << "  Address:       " << found->getAddress() << "\n";
                    std::cout << "  Course:        " << found->getCourseId() << "\n";
                    std::cout << "  Status:        " << found->getStatus() << "\n";
                    printDivider();
                } else {
                    std::cout << "\n[INFO] No matching record found for ID '" << queryId << "'.\n";
                }
            } else if (op == 4) {
                std::vector<Course> courses = repo.getAllCourses();
                std::cout << "\n--- AVAILABLE COURSES (" << courses.size() << ") ---\n";
                std::cout << std::left << std::setw(10) << "Course ID"
                          << std::setw(35) << "Course Name"
                          << std::setw(15) << "Department"
                          << std::setw(10) << "Duration" << "\n";
                printDivider();
                for (const auto& c : courses) {
                    std::cout << std::left << std::setw(10) << c.getCourseId()
                              << std::setw(35) << c.getCourseName()
                              << std::setw(15) << c.getDepartment()
                              << c.getDurationYears() << " Years\n";
                }
                printDivider();
            } else {
                std::cout << "\n[ERROR] Invalid option selected. Please try again.\n";
            }
        }
    }

    return 0;
}
